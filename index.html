async function sendMessage(){ if(busy)return; const message=$('chatInput').value.trim(); if(!message)return; busy=true; $('sendChat').disabled=true; $('chatStatus').textContent='Gemini está preparando una respuesta…'; addBubble('user',message); $('chatInput').value=''; history.push({role:'user',parts:[{text:message}]}); const wait=addBubble('ai','Analizando…'); try{ const conversation=history.map(item=>{ const role=item.role==='model'?'AGENTE':'USUARIO'; const text=(item.parts||[]).map(part=>part.text||'').join('\n'); return role+': '+text; }).join('\n\n'); const prompt=systemInstruction+'\n\n'+contextText()+ '\n\nHISTORIAL DE CONVERSACIÓN:\n'+conversation+ '\n\nResponde a la última consulta del usuario.'; const response=await fetch( 'https://gemelo-digital-estrategico.vercel.app/api/chat', { method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({prompt}) } ); const data=await response.json(); if(!response.ok){ throw new Error(data?.error||`Error HTTP ${response.status}`); } const answer=typeof data.text==='string'?data.text.trim():''; if(!answer){ throw new Error('El servidor no devolvió texto.'); } wait.textContent=answer; history.push({role:'model',parts:[{text:answer}]}); $('chatStatus').textContent='Respuesta recibida.'; }catch(err){ wait.textContent='No se pudo conectar: '+err.message; history.pop(); $('chatStatus').textContent='Conexión no completada.'; }finally{ busy=false; $('sendChat').disabled=false; $('chatInput').focus(); } }
